        // Inicializar Firebase

        // Revisar si ya hay sesión activa
        auth.onAuthStateChanged(user => {

        function setLoginMethod(method) {
            selectedMethod = method;
            document.getElementById('emailFields').classList.toggle('hidden', method !== 'Correo Electrónico');
            
            // Resaltar botón activo
            document.getElementById('btnEmail').style.borderColor = method === 'Correo Electrónico' ? '#db2777' : '#fbcfe8';

        function mostrarError(mensaje) {
            const el = document.getElementById('errorMsg');
            el.textContent = '❌ ' + mensaje;
            el.classList.remove('hidden');
            setTimeout(() => el.classList.add('hidden'), 5000);
        }

        async function handleRegistration() {
            const wp = document.getElementById('gateWhatsapp').value.trim().replace(/\D/g, '');
            const email = document.getElementById('userEmail').value.trim();
            const pass = document.getElementById('userPass').value;

            if (wp.length < 10) {
                mostrarError('Ingresa un número de WhatsApp válido (10 dígitos)');
                return;
            }

            try {
                document.getElementById('registerBtn').disabled = true;
                document.getElementById('registerBtn').textContent = 'Verificando...';

                let userCredential;

                if (selectedMethod === 'Correo Electrónico') {
                    if (!email || pass.length < 6) {
                        mostrarError('Correo inválido o contraseña debe tener al menos 6 caracteres');
                        return;
                    }
                    try {
                        userCredential = await auth.createUserWithEmailAndPassword(email, pass);
                    } catch (e) {
                        if (e.code === 'auth/email-already-in-use') {
                            userCredential = await auth.signInWithEmailAndPassword(email, pass);
                        } else {
                            throw e;
                        }
                    }
                } else if (selectedMethod === 'Facebook') {
                    const provider = new firebase.auth.FacebookAuthProvider();
                    userCredential = await auth.signInWithPopup(provider);
                } else if (selectedMethod === 'ID de Cliente') {
                    const tempEmail = `user_${wp}@clickandcut.mx`;
                    const tempPass = wp + 'Abc!';
                    try {
                        userCredential = await auth.signInWithEmailAndPassword(tempEmail, tempPass);
                    } catch (e) {
                        if (e.code === 'auth/user-not-found') {
                            userCredential = await auth.createUserWithEmailAndPassword(tempEmail, tempPass);
                        } else {
                            throw e;
                        }
                    }
                }

                // Guardar datos del cliente en Firestore
                const user = userCredential.user;
                await db.collection('clientes').doc(user.uid).set({
                    metodo: selectedMethod,
                    whatsapp: wp,
                    email: user.email || '',
                    fechaRegistro: firebase.firestore.FieldValue.serverTimestamp()
                }, { merge: true });

                mostrarTienda(user);

            } catch (error) {
                console.error(error);
                mostrarError(error.message);
            } finally {
                document.getElementById('registerBtn').disabled = false;
                document.getElementById('registerBtn').textContent = 'Entrar a la Tienda 🩷';
            }
        }

        function mostrarTienda(user) {
            document.getElementById('gateModal').classList.add('hidden');
            document.getElementById('mainApp').classList.remove('hidden');
            document.getElementById('userBadge').textContent = user.email ? user.email.split('@')[0] : 'Cliente Verificado';
        }

        function cerrarSesion() {
            auth.signOut().then(() => {
                document.getElementById('mainApp').classList.add('hidden');
                document.getElementById('gateModal').classList.remove('hidden');
            });
        }

        function selectProduct(product, price) {
            const phone = "528641114514";
            const text = encodeURIComponent(`¡Hola! 🩷 Me interesa *${product}* (${price}). ¿Me das más información?`);
            window.open(`https://wa.me/${phone}?text=${text}`, '_blank');
        }

        function searchApps() {
            const term = document.getElementById('searchInput').value.toLowerCase();
            document.querySelectorAll('.app-card').forEach(card => {
                const name = card.getAttribute('data-name');
                card.style.display = name.includes(term) ? 'flex' : 'none';
            });
        }
